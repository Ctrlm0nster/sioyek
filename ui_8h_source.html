665 QStringList get_dir_contents(QString root, QString prefix) {
666
667 root = expand_home_dir(root);
668 QDir directory(root);
669 QFileInfoList entry_list = directory.entryInfoList(
670 { prefix + "*" },
671 QDir::NoDot|QDir::Dirs|QDir::Files|QDir::Readable
672 );
673 QStringList result {};
674 QString sep = QDir::separator();
675
676 for (const auto& entry : entry_list) {
677 if (entry.isDir()) {
678 result.push_back(entry.fileName() + sep);
679 }
680 else {
681 result.push_back(entry.fileName());
682 }
683 }
684 if (result.size() == 0) {
685 const std::string encoded_prefix = utf8_encode(prefix.toStdWString());
686 const QFileInfoList all_directory_files = directory.entryInfoList();
687 std::vector<std::pair<QString, int>> file_scores;
688
689 for (const auto& file : all_directory_files) {
690 std::string encoded_file = utf8_encode(file.fileName().toStdWString());
691 int score = 0;
692 if (is_fuzzy) {
693 score = static_cast<int>(rapidfuzz::fuzz::partial_ratio(encoded_prefix, encoded_file));
694 }
695 else {
696 fts::fuzzy_match(encoded_prefix.c_str(), encoded_file.c_str(), score);
697 }
698 const auto filename = (file.isDir() ? (file.fileName() + sep) : file.fileName());
699 file_scores.push_back(std::make_pair(filename, score));
700 }
701 std::sort(file_scores.begin(), file_scores.end(), [](std::pair<QString, int> lhs, std::pair<QString, int> rhs) {
702 return lhs.second > rhs.second;
703 });
704 for (const auto& [file, score] : file_scores) {
705 if (score > 0) {
706 result.push_back(file);
707 }
708 }
709 }
710 return result;
