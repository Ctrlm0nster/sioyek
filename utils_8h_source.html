214 for (int j = 0; j < d3; j++) {
215 result[i * d3 + j] = 0;

381template<typename R>
382void merge_selected_character_rects(const std::deque<R>& selected_character_rects, std::vector<R>& resulting_rects, bool touch_vertically=true) {
383 /*
384 This function merges the bounding boxes of all selected characters into large line chunks.
385 */
386
387 if (selected_character_rects.size() == 0) {
388 return;
389 }
390
391 std::vector<fz_rect> line_rects;
392
393 fz_rect last_rect = selected_character_rects[0];
394 line_rects.push_back(selected_character_rects[0]);
395
396 for (size_t i = 1; i < selected_character_rects.size(); i++) {
397 if (is_consequtive(last_rect, selected_character_rects[i])) {
398 last_rect = selected_character_rects[i];
399 line_rects.push_back(selected_character_rects[i]);
400 }
401 else {
402 fz_rect bounding_rect = bound_rects(line_rects);
403 resulting_rects.push_back(bounding_rect);
404 line_rects.clear();
405 last_rect = selected_character_rects[i];
406 line_rects.push_back(selected_character_rects[i]);
407 }
408 }
409
410 if (line_rects.size() > 0) {
411 fz_rect bounding_rect = bound_rects(line_rects);
412 resulting_rects.push_back(bounding_rect);
413 }
414
415 // avoid overlapping rects
416 for (size_t i = 0; i < resulting_rects.size() - 1; i++) {
417 // we don't need to do this across columns of document
418 float height = std::abs(resulting_rects[i].y1 - resulting_rects[i].y0);
419 if (std::abs(resulting_rects[i + 1].y0 - resulting_rects[i].y0) < (0.5 * height)) {
420 continue;
421 }
422 if (touch_vertically) {
423 if ((resulting_rects[i + 1].x0 < resulting_rects[i].x1)) {
424 const float MERGING_LINE_DISTANCE_THRESHOLD = 25.0f;
425 if (std::abs(resulting_rects[i + 1].y0 - resulting_rects[i].y1) < MERGING_LINE_DISTANCE_THRESHOLD) {
426 resulting_rects[i + 1].y0 = resulting_rects[i].y1;
427 }
428 }
429 }
430 }
431
432}
