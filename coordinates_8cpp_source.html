14bool range_intersects(float range1_start, float range1_end, float range2_start, float range2_end) {
15 if (range1_start > range1_end) {
16 std::swap(range1_start, range1_end);
17 }
18 if (range2_start > range2_end) {
19 std::swap(range2_start, range2_end);
20 }
21 if (range2_start > range1_end || range1_start > range2_end) {
22 return false;
23 }
24 return true;
25}
