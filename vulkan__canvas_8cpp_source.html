84QShader loadShader(const char* path) {

86 if (!file.open(QIODevice::ReadOnly)) qFatal("Cannot open embedded Vulkan shader %s", path);
87 auto shader = QShader::fromSerialized(file.readAll());
88 if (!shader.isValid()) qFatal("Invalid embedded Vulkan shader %s", path);
89 const auto variants = shader.availableShaders();
90 if (variants.isEmpty() || std::any_of(variants.begin(), variants.end(), [](const QShaderKey& key) {

224 impl->logical_size = size();

230 impl->painter.reset();

233 const int alignment = rhi()->ubufAligned(sizeof(ShaderUniforms));
234 std::size_t total_vertices = 0;
235 for (const auto& draw : impl->draws) total_vertices += draw.vertices.size();
236 const auto vertex_bytes = total_vertices * sizeof(Vertex);
237 const auto uniform_bytes = impl->draws.size() * alignment;
238 if (vertex_bytes > std::size_t(std::numeric_limits<int>::max()) ||
239 uniform_bytes > std::size_t(std::numeric_limits<int>::max()))
240 qFatal("Frame geometry exceeds Vulkan buffer limits");
241 auto ensureBuffer = [&](auto& buffer, int& capacity, int needed, QRhiBuffer::UsageFlag usage) {
242 if (capacity >= needed && buffer) return;
243 capacity = std::max(needed, 4096);
244 buffer.reset(rhi()->newBuffer(QRhiBuffer::Dynamic, usage, capacity));
245 createResource(buffer);
246 };
247 if (int(uniform_bytes) > impl->uniform_capacity) {
248 // Bindings reference the old uniform buffer; invalidate them before replacement.
249 for (auto& [id, texture] : impl->textures) texture.bindings.reset();
250 impl->pipelines.clear();
251 }
252 ensureBuffer(impl->vertex_buffer, impl->vertex_capacity, int(vertex_bytes), QRhiBuffer::VertexBuffer);
253 ensureBuffer(impl->uniform_buffer, impl->uniform_capacity, int(uniform_bytes), QRhiBuffer::UniformBuffer);
254 if (!impl->sampler || impl->linear_filtering != LINEAR_TEXTURE_FILTERING) {
255 for (auto& [id, texture] : impl->textures) texture.bindings.reset();
256 impl->linear_filtering = LINEAR_TEXTURE_FILTERING;
257 const auto filter = LINEAR_TEXTURE_FILTERING ? QRhiSampler::Linear : QRhiSampler::Nearest;
258 impl->sampler.reset(rhi()->newSampler(filter, filter, QRhiSampler::None,
259 QRhiSampler::ClampToEdge, QRhiSampler::ClampToEdge));
260 createResource(impl->sampler);
261 }
262 // Device-local textures belong to this widget/device. Retain only live CPU images.
263 for (auto it = impl->textures.begin(); it != impl->textures.end();) {
264 if (it->first && !RenderImages::contains(it->first)) it = impl->textures.erase(it);
265 else ++it;
266 }
267 auto* updates = rhi()->nextResourceUpdateBatch();
268 auto bindingsFor = [&](RenderHandle id) -> QRhiShaderResourceBindings* {
269 auto& entry = impl->textures[id];
270 if (!entry.texture) {
271 QImage image;
272 if (id) image = RenderImages::get(id);
273 else {
274 image = QImage(1, 1, QImage::Format_RGBA8888);
275 image.fill(Qt::white);
276 }
277 if (image.isNull()) qFatal("Missing cached page image");
278 // Keep premultiplied painter pixels: overlay blending uses One.
279 if (image.format() == QImage::Format_RGBA8888_Premultiplied)
280 image.reinterpretAsFormat(QImage::Format_RGBA8888);
281 else image = image.convertToFormat(QImage::Format_RGBA8888);
282 entry.texture.reset(rhi()->newTexture(QRhiTexture::RGBA8, image.size()));
283 createResource(entry.texture);
284 updates->uploadTexture(entry.texture.get(), image);
285 }
286 if (!entry.bindings) {
287 entry.bindings.reset(rhi()->newShaderResourceBindings());
288 entry.bindings->setBindings({
289 QRhiShaderResourceBinding::uniformBufferWithDynamicOffset(
290 0, QRhiShaderResourceBinding::VertexStage | QRhiShaderResourceBinding::FragmentStage,
291 impl->uniform_buffer.get(), sizeof(ShaderUniforms)),
292 QRhiShaderResourceBinding::sampledTexture(
293 1, QRhiShaderResourceBinding::FragmentStage, entry.texture.get(), impl->sampler.get())
294 });
295 createResource(entry.bindings);
296 }
297 return entry.bindings.get();
298 };
299 std::vector<QRhiShaderResourceBindings*> bindings;
300 std::vector<QRhiGraphicsPipeline*> pipelines;
301 bindings.reserve(impl->draws.size());
302 pipelines.reserve(impl->draws.size());
303 quint32 vertex_offset = 0, uniform_offset = 0;
304 const QMatrix4x4 correction = rhi()->clipSpaceCorrMatrix();
305 for (auto& draw : impl->draws) {
306 std::memcpy(draw.uniforms.correction, correction.constData(), 64);
307 const auto bytes = quint32(draw.vertices.size() * sizeof(Vertex));
308 updates->updateDynamicBuffer(impl->vertex_buffer.get(), vertex_offset, bytes, draw.vertices.data());
309 updates->updateDynamicBuffer(impl->uniform_buffer.get(), uniform_offset, sizeof(ShaderUniforms), &draw.uniforms);
310 vertex_offset += bytes;
311 uniform_offset += alignment;
312 auto* binding = bindingsFor(draw.image);
313 bindings.push_back(binding);
314 auto& pipeline = impl->pipelines[draw.pipeline];
315 if (!pipeline) {
316 pipeline.reset(rhi()->newGraphicsPipeline());
317 const auto& key = draw.pipeline;
318 pipeline->setShaderStages({{QRhiShaderStage::Vertex, impl->vertex_shader},
319 {QRhiShaderStage::Fragment, impl->fragment_shader}});
320 QRhiVertexInputLayout input;
321 input.setBindings({{sizeof(Vertex)}});
322 input.setAttributes({{0, 0, QRhiVertexInputAttribute::Float2, offsetof(Vertex, position)},
323 {0, 1, QRhiVertexInputAttribute::Float2, offsetof(Vertex, uv)},
324 {0, 2, QRhiVertexInputAttribute::Float4, offsetof(Vertex, color)}});
325 pipeline->setVertexInputLayout(input);
326 pipeline->setTopology(key.lines ? QRhiGraphicsPipeline::Lines : QRhiGraphicsPipeline::Triangles);
327 pipeline->setCullMode(QRhiGraphicsPipeline::None);
328 pipeline->setSampleCount(renderTarget()->sampleCount());
329 QRhiGraphicsPipeline::TargetBlend blend;
330 blend.enable = key.blend;
331 blend.colorWrite = QRhiGraphicsPipeline::ColorMask(key.color_mask);
332 blend.srcColor = blendFactor(key.src);
333 blend.dstColor = blendFactor(key.dst);
334 blend.srcAlpha = blendFactor(key.src_alpha);
335 blend.dstAlpha = blendFactor(key.dst_alpha);
336 pipeline->setTargetBlends({blend});
337 pipeline->setDepthTest(key.depth || key.clear_depth);
338 pipeline->setDepthWrite(key.depth || key.clear_depth);
339 pipeline->setDepthOp(key.clear_depth ? QRhiGraphicsPipeline::Always : QRhiGraphicsPipeline::Less);
340 pipeline->setStencilTest(key.stencil);
341 QRhiGraphicsPipeline::StencilOpState stencil;
342 stencil.compareOp = compareOp(key.compare);
343 stencil.failOp = stencilOperation(key.fail);
344 stencil.depthFailOp = stencilOperation(key.depth_fail);
345 stencil.passOp = stencilOperation(key.pass);
346 pipeline->setStencilFront(stencil);
347 pipeline->setStencilBack(stencil);
348 pipeline->setStencilReadMask(key.stencil_read);
349 pipeline->setStencilWriteMask(key.stencil_write);
350 pipeline->setFlags(QRhiGraphicsPipeline::UsesStencilRef | QRhiGraphicsPipeline::UsesScissor);
351 pipeline->setShaderResourceBindings(binding);
352 pipeline->setRenderPassDescriptor(renderTarget()->renderPassDescriptor());
353 createResource(pipeline);
354 }
355 pipelines.push_back(pipeline.get());
356 }
357 cb->beginPass(renderTarget(), impl->background, {1.0f, 0}, updates);
358 vertex_offset = uniform_offset = 0;
359 for (std::size_t i = 0; i < impl->draws.size(); ++i) {
360 const auto& draw = impl->draws[i];
361 cb->setGraphicsPipeline(pipelines[i]);
362 cb->setViewport(QRhiViewport(0, 0, pixel_size.width(), pixel_size.height()));
363 cb->setScissor(QRhiScissor(0, 0, pixel_size.width(), pixel_size.height()));
364 const QRhiCommandBuffer::DynamicOffset offset(0, uniform_offset);
365 cb->setShaderResources(bindings[i], 1, &offset);
366 cb->setStencilRef(draw.stencil_ref);
367 const QRhiCommandBuffer::VertexInput input(impl->vertex_buffer.get(), vertex_offset);
368 cb->setVertexInput(0, 1, &input);
369 cb->draw(quint32(draw.vertices.size()));
370 vertex_offset += quint32(draw.vertices.size() * sizeof(Vertex));
371 uniform_offset += alignment;
372 }
373 cb->endPass();
374 for (const auto image : impl->overlay_images) RenderImages::remove(image);
375 impl->overlay_images.clear();
376}

394void VulkanCanvas::flushPainter() {
395 if (!impl->painter_dirty) return;
396 impl->painter_dirty = false;
397 // Rasterized labels/icons are composed at their original position in the
398 // command stream, between GPU page/highlight/drawing passes. Most PDF
399 // frames contain no painter content: do not copy/upload transparent layers.
400 const QImage& overlay = impl->overlay;
401 int left = overlay.width(), top = overlay.height(), right = -1, bottom = -1;
402 constexpr quint32 alpha_mask = Q_BYTE_ORDER == Q_LITTLE_ENDIAN ? 0xff000000u : 0x000000ffu;
403 for (int y = 0; y < overlay.height(); ++y) {
404 const auto* row = reinterpret_cast<const quint32*>(overlay.constScanLine(y));
405 const auto* end = row + overlay.width();
406 const auto* first = std::find_if(row, end, [](quint32 pixel) { return pixel & alpha_mask; });
407 if (first == end) continue;
408 const auto last = std::find_if(std::make_reverse_iterator(end),
409 std::make_reverse_iterator(first),
410 [](quint32 pixel) { return pixel & alpha_mask; });
411 left = std::min(left, int(first - row));
412 right = std::max(right, int(last.base() - row) - 1);
413 top = std::min(top, y);
414 bottom = y;
415 }
416 if (right < left) return;
417
418 const QRect bounds(QPoint(left, top), QPoint(right, bottom));
419 QImage pixels = overlay.copy(bounds);
420 // This is an upload-only image. Change the tag while the pixels are unique;
421 // reinterpreting a shared RenderImages entry would detach/copy them again.
422 pixels.reinterpretAsFormat(QImage::Format_RGBA8888);
423 Draw draw;
424 draw.pipeline.blend = true;
425 draw.pipeline.src = draw.pipeline.src_alpha = Gpu::One;
426 draw.pipeline.dst = draw.pipeline.dst_alpha = Gpu::OneMinusSrcAlpha;
427 draw.uniforms.params[0] = Gpu::Overlay;
428 draw.image = RenderImages::insert(pixels);
429 impl->overlay_images.push_back(draw.image);
430 // Bounds are physical pixels; use the full render-target size for NDC,
431 // independent of logical widget size and fractional display scaling.
432 const float x0 = 2.0f * left / overlay.width() - 1.0f;
433 const float x1 = 2.0f * (right + 1) / overlay.width() - 1.0f;
434 const float y0 = 1.0f - 2.0f * top / overlay.height();
435 const float y1 = 1.0f - 2.0f * (bottom + 1) / overlay.height();
436 draw.vertices = {{{x0, y0}, {0, 0}}, {{x0, y1}, {0, 1}}, {{x1, y0}, {1, 0}},
437 {{x1, y0}, {1, 0}}, {{x0, y1}, {0, 1}}, {{x1, y1}, {1, 1}}};
438 impl->draws.push_back(std::move(draw));
439 // The rest of the image is already transparent. Keep painter state intact
440 // across GPU boundaries while clearing only the pixels just composed.
441 for (int y = top; y <= bottom; ++y)
442 std::memset(impl->overlay.scanLine(y) + left * 4, 0, bounds.width() * 4);
443}

573void VulkanCanvas::drawElements(Gpu::Value topology, int count, Gpu::Value type, const void* offset) {
574 if (type != Gpu::UnsignedInt || count < 0) qFatal("Invalid index format");
575 if (count == 0) return;
576 const auto& bytes = impl->buffers.at(impl->currentLayout().index_buffer);
577 const auto start = reinterpret_cast<std::uintptr_t>(offset);
578 const auto size = std::size_t(count) * sizeof(std::uint32_t);
579 if (start > std::size_t(bytes.size()) || size > std::size_t(bytes.size()) - start)
580 qFatal("Index range exceeds geometry buffer");

582 std::memcpy(indices.data(), bytes.constData() + start, size);
583 recordDraw(topology, indices);

586void VulkanCanvas::recordDraw(Gpu::Value topology, const std::vector<std::uint32_t>& indices) {
587 Draw draw;
588 draw.pipeline = impl->state;
589 draw.pipeline.lines = topology == Gpu::Lines || topology == Gpu::LineLoop;
590 draw.stencil_ref = impl->stencil_ref;
591 const auto& uniforms = impl->uniforms();
592 std::copy_n(uniforms.transform.constData(), 16, draw.uniforms.transform);
593 std::copy(uniforms.color.begin(), uniforms.color.end(), draw.uniforms.color);
594 draw.uniforms.params[0] = impl->program;
595 draw.uniforms.params[1] = uniforms.contrast;
596 if (impl->program == Gpu::Page || impl->program == Gpu::DarkPage || impl->program == Gpu::CustomPage)
597 draw.image = impl->texture;
598 const auto& attributes = impl->currentLayout().attributes;
599 auto vertex = [&](std::uint32_t index) {
600 Vertex result;
601 for (int k = 0; k < 2; ++k) result.position[k] = impl->component(attributes[0], index, k);
602 if (impl->program == Gpu::CompiledLine || impl->program == Gpu::CompiledDot) {
603 result.position[0] = result.position[0] * uniforms.scale[0] + uniforms.offset[0];
604 result.position[1] = -result.position[1] * uniforms.scale[1] + uniforms.offset[1];
605 const int type = int(impl->component(attributes[impl->program == Gpu::CompiledLine ? 1 : 2], index, 0));
606 if (type < 0 || type >= 26) qFatal("Invalid freehand drawing color index");
607 std::copy_n(uniforms.type_colors.data() + type * 3, 3, result.color);
608 }
609 const bool needs_uv = impl->program == Gpu::Page || impl->program == Gpu::DarkPage ||
610 impl->program == Gpu::CustomPage || impl->program == Gpu::Placeholder ||
611 impl->program == Gpu::CompiledDot;
612 if (needs_uv && attributes[1].components == 2) {
613 for (int k = 0; k < 2; ++k) result.uv[k] = impl->component(attributes[1], index, k);
614 }
615 return result;
616 };
617 // Expand fans/strips/primitive restarts into portable triangle/line lists.
618 // This preserves the original geometry on drivers without triangle fans.
619 std::vector<std::uint32_t> segment;
620 auto emitSegment = [&] {
621 auto appendVertex = [&](std::size_t i) { draw.vertices.push_back(vertex(segment[i])); };
622 const auto n = segment.size();
623 if (topology == Gpu::TriangleStrip || topology == Gpu::TriangleFan) {
624 for (std::size_t i = 2; i < n; ++i) {
625 appendVertex(topology == Gpu::TriangleFan ? 0 : i - 2);
626 appendVertex(i - 1); appendVertex(i);
627 }
628 } else if (topology == Gpu::LineLoop) {
629 for (std::size_t i = 0; i < n; ++i) { appendVertex(i); appendVertex((i + 1) % n); }
630 } else {
631 for (std::size_t i = 0; i < n; ++i) appendVertex(i);
632 }
633 segment.clear();
634 };
635 for (auto index : indices) {
636 if (index == std::numeric_limits<std::uint32_t>::max()) emitSegment();
637 else segment.push_back(index);
638 }
639 emitSegment();
640 if (!draw.vertices.empty()) impl->draws.push_back(std::move(draw));
641}
