73 template <typename octet_iterator>
74 octet_iterator append(uint32_t cp, octet_iterator result)
75 {
76 if (!utf8::internal::is_code_point_valid(cp))
77 throw invalid_code_point(cp);
78
79 if (cp < 0x80) // one octet
80 *(result++) = static_cast<uint8_t>(cp);
81 else if (cp < 0x800) { // two octets
82 *(result++) = static_cast<uint8_t>((cp >> 6) | 0xc0);
83 *(result++) = static_cast<uint8_t>((cp & 0x3f) | 0x80);
84 }
85 else if (cp < 0x10000) { // three octets
86 *(result++) = static_cast<uint8_t>((cp >> 12) | 0xe0);
87 *(result++) = static_cast<uint8_t>(((cp >> 6) & 0x3f) | 0x80);
88 *(result++) = static_cast<uint8_t>((cp & 0x3f) | 0x80);
89 }
90 else { // four octets
91 *(result++) = static_cast<uint8_t>((cp >> 18) | 0xf0);
92 *(result++) = static_cast<uint8_t>(((cp >> 12) & 0x3f) | 0x80);
93 *(result++) = static_cast<uint8_t>(((cp >> 6) & 0x3f) | 0x80);
94 *(result++) = static_cast<uint8_t>((cp & 0x3f) | 0x80);
95 }
96 return result;
97 }

99 template <typename octet_iterator, typename output_iterator>
100 output_iterator replace_invalid(octet_iterator start, octet_iterator end, output_iterator out, uint32_t replacement)
101 {
102 while (start != end) {
103 octet_iterator sequence_start = start;
104 internal::utf_error err_code = utf8::internal::validate_next(start, end);
105 switch (err_code) {
106 case internal::UTF8_OK :
107 for (octet_iterator it = sequence_start; it != start; ++it)
108 *out++ = *it;
109 break;
110 case internal::NOT_ENOUGH_ROOM:
111 throw not_enough_room();
112 case internal::INVALID_LEAD:
113 out = utf8::append (replacement, out);
114 ++start;
115 break;
116 case internal::INCOMPLETE_SEQUENCE:
117 case internal::OVERLONG_SEQUENCE:
118 case internal::INVALID_CODE_POINT:
119 out = utf8::append (replacement, out);
120 ++start;
121 // just one replacement mark for the sequence
122 while (start != end && utf8::internal::is_trail(*start))
123 ++start;
124 break;
125 }
126 }
127 return out;
128 }

137 template <typename octet_iterator>
138 uint32_t next(octet_iterator& it, octet_iterator end)
139 {
140 uint32_t cp = 0;
141 internal::utf_error err_code = utf8::internal::validate_next(it, end, cp);
142 switch (err_code) {
143 case internal::UTF8_OK :
144 break;
145 case internal::NOT_ENOUGH_ROOM :
146 throw not_enough_room();
147 case internal::INVALID_LEAD :
148 case internal::INCOMPLETE_SEQUENCE :
149 case internal::OVERLONG_SEQUENCE :
150 throw invalid_utf8(*it);
151 case internal::INVALID_CODE_POINT :
152 throw invalid_code_point(cp);
153 }
154 return cp;
155 }

163 template <typename octet_iterator>
164 uint32_t prior(octet_iterator& it, octet_iterator start)
165 {
166 // can't do much if it == start
167 if (it == start)
168 throw not_enough_room();
169
170 octet_iterator end = it;
171 // Go back until we hit either a lead octet or start
172 while (utf8::internal::is_trail(*(--it)))
173 if (it == start)
174 throw invalid_utf8(*it); // error - no lead byte in the sequence
175 return utf8::peek_next(it, end);
176 }

179 template <typename octet_iterator>
180 uint32_t previous(octet_iterator& it, octet_iterator pass_start)
181 {
182 octet_iterator end = it;
183 while (utf8::internal::is_trail(*(--it)))
184 if (it == pass_start)
185 throw invalid_utf8(*it); // error - no lead byte in the sequence
186 octet_iterator temp = it;
187 return utf8::next(temp, end);

207 template <typename u16bit_iterator, typename octet_iterator>
208 octet_iterator utf16to8 (u16bit_iterator start, u16bit_iterator end, octet_iterator result)
209 {
210 while (start != end) {
211 uint32_t cp = utf8::internal::mask16(*start++);
212 // Take care of surrogate pairs first
213 if (utf8::internal::is_lead_surrogate(cp)) {
214 if (start != end) {
215 uint32_t trail_surrogate = utf8::internal::mask16(*start++);
216 if (utf8::internal::is_trail_surrogate(trail_surrogate))
217 cp = (cp << 10) + trail_surrogate + internal::SURROGATE_OFFSET;
218 else
219 throw invalid_utf16(static_cast<uint16_t>(trail_surrogate));
220 }
221 else
222 throw invalid_utf16(static_cast<uint16_t>(cp));
223
224 }
225 // Lone trail surrogate
226 else if (utf8::internal::is_trail_surrogate(cp))
227 throw invalid_utf16(static_cast<uint16_t>(cp));
228
229 result = utf8::append(cp, result);
230 }
231 return result;
232 }

234 template <typename u16bit_iterator, typename octet_iterator>
235 u16bit_iterator utf8to16 (octet_iterator start, octet_iterator end, u16bit_iterator result)
236 {
237 while (start != end) {
238 uint32_t cp = utf8::next(start, end);
239 if (cp > 0xffff) { //make a surrogate pair
240 *result++ = static_cast<uint16_t>((cp >> 10) + internal::LEAD_OFFSET);
241 *result++ = static_cast<uint16_t>((cp & 0x3ff) + internal::TRAIL_SURROGATE_MIN);
242 }
243 else
244 *result++ = static_cast<uint16_t>(cp);
245 }
246 return result;
247 }
