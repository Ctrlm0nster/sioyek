713float Document::get_page_size_smart(bool width, int page_index, float* left_ratio, float* right_ratio, int* normal_page_width) {
714
715 fz_pixmap* pixmap = get_small_pixmap(page_index);
716
717 // project the small pixmap into a histogram (the `width` parameter determines whether we do this
718 // horizontally or vertically)
719 std::vector<float> histogram;
720
721 if (width) {
722 histogram.resize(pixmap->w);
723 }
724 else {
725 histogram.resize(pixmap->h);
726 }
727
728 float total_nonzero = 0.0f;
729 for (int i = 0; i < pixmap->w; i++) {
730 for (int j = 0; j < pixmap->h; j++) {
731 int index = 3 * pixmap->w * j + 3 * i;
732
733 int r = pixmap->samples[index];
734 int g = pixmap->samples[index + 1];
735 int b = pixmap->samples[index + 2];
736
737 float brightness = static_cast<float>(r + g + b) / 3.0f;
738 if (brightness < 250) {
739 if (width) {
740 histogram[i] += 1;
741 }
742 else {
743 histogram[j] += 1;
744 }
745 total_nonzero += 1;
746 }
747 }
748 }
749
750 float average_nonzero = total_nonzero / histogram.size();
751 float nonzero_threshold = average_nonzero / 3.0f;
752
753 int start_index = 0;
754 int end_index = histogram.size() - 1;
755
756 // find the first index in both directions where the histogram is nonzero
757 while ((start_index < end_index) && (histogram[start_index++] < nonzero_threshold));
758 while ((start_index < end_index) && (histogram[end_index--] < nonzero_threshold));
759
760
761 int border = 10;
762 start_index = std::max<int>(start_index - border, 0);
763 end_index = std::min<int>(end_index + border, histogram.size() - 1);
764
765 *left_ratio = static_cast<float>(start_index) / histogram.size();
766 *right_ratio = static_cast<float>(end_index) / histogram.size();
767
768 float standard_size;
769 if (width) {
770 standard_size = page_widths[page_index];
771 }
772 else {
773 standard_size = page_heights[page_index];
774 }
775
776 float ratio = static_cast<float>(end_index - start_index) / histogram.size();
777
778 *normal_page_width = standard_size;
779
780 return ratio * standard_size;
781}
