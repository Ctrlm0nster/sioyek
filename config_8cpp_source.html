440 stream >> res->enabled >> res->left >> res->right >> res->top >> res->bottom;
441 if (changed) {
442 UIRect new_value = *res;
443 if ((prev_value.bottom != new_value.bottom) ||
444 (prev_value.top != new_value.top) ||
445 (prev_value.left != new_value.left) ||
446 (prev_value.right != new_value.right) ||
447 (prev_value.enabled != new_value.enabled)) {
448 *changed = true;
449 }
450 else {
451 *changed = false;
452 }
453
454 }
455
456 return res;
457

470template<int N, typename T>
471void* vec_n_deserializer(std::wstringstream& stream, void* res_, bool* changed) {
472 assert(res_ != nullptr);
473 T* res = (T*)res_;
474 T prev_value[N];
475 for (int i = 0; i < N; i++) {
476 prev_value[i] = *(res + i);
477 }
478
479 if (res == nullptr) {
480 res = new T[N];
481 assert(false); // this should not happen
482 }
483 for (int i = 0; i < N; i++) {
484 stream >> *(res + i);
485 }
486 if (changed) {
487 *changed = are_vecs_different<N, T>(prev_value, res);
488
489 }
490
491 return res;

496
497 assert(res_ != nullptr);
498 float* res = (float*)res_;
499 float prev_value[N];
500 for (int i = 0; i < N; i++) {
501 prev_value[i] = *(res + i);
502 }
503
504 if (res == nullptr) {
505 res = new float[N];
506 assert(false); // this should not happen
507 }
508 while (std::isspace(stream.peek())) {
509 stream.ignore();
510 }
511 if (stream.peek() == '#') {
512 std::wstring rest;
513 std::getline(stream, rest);
514 hexademical_to_normalized_color(rest, res, N);
515 }
516 else {
517 for (int i = 0; i < N; i++) {
518 stream >> *(res + i);
519 }
520 }
521 if (changed) {
522 if (are_vecs_different<N, float>(prev_value, res)) {
523 *changed = true;
524 }
525 else {
526 *changed = false;
527 }
528 }
529
530 return res;

1221void ConfigManager::deserialize_file(std::vector<std::string>* changed_config_names, const Path& file_path, bool warn_if_not_exists, bool is_auto) {
1222
1223 std::wstring line;
1224 std::wifstream default_file = open_wifstream(file_path.get_path());
1225 int line_number = 0;
1226
1227 if (warn_if_not_exists && (!default_file.good())) {
1228 std::wcout << "Error: Could not open config file " << file_path.get_path() << std::endl;
1229 }
1230
1231 while (std::getline(default_file, line)) {
1232 line_number++;
1233
1234 if (line.size() == 0 || line[0] == '#') {
1235 continue;
1236 }
1237
1238 std::wstringstream ss{ line };
1239 std::wstring conf_name;
1240 ss >> conf_name;
1241 //special handling for new_command
1242 if (conf_name == L"source") {
1243 std::wstring path;
1244 std::getline(ss, path);
1245 path = strip_string(path);
1246 if (path.size() > 0) {
1247 if (path[0] == '.') {
1248 auto parent_dir = QDir(QString::fromStdWString(file_path.file_parent().get_path()));
1249 path = parent_dir.absoluteFilePath(QString::fromStdWString(path)).toStdWString();
1250 }
1251 if (path[0] == '~') {
1252 path = QDir::homePath().toStdWString() + path.substr(1, path.size() - 1);
1253 }
1254
1255 deserialize_file(changed_config_names, path, true);
1256 }
1257 }
1258 else if ((conf_name == L"new_command") || (conf_name == L"new_async_js_command") || (conf_name == L"new_macro") || (conf_name == L"new_js_command") || (conf_name == L"keymap")) {
1259 std::wstring config_value;
1260 std::getline(ss, config_value);
1261 config_value = strip_string(config_value);
1262 int space_index = config_value.find(L" ");
1263
1264 if (conf_name == L"keymap") {
1265 AdditionalKeymapData keymap_data;
1266 keymap_data.file_name = file_path.get_path();
1267 keymap_data.line_number = line_number;
1268 keymap_data.keymap_string = QString::fromStdWString(config_value).trimmed().toStdWString();
1269 ADDITIONAL_KEYMAPS.push_back(keymap_data);
1270 }
1271 else {
1272 std::wstring new_command_name = config_value.substr(0, space_index);
1273 if (new_command_name[0] == '_') {
1274 std::wstring command_value = config_value.substr(space_index + 1, config_value.size() - space_index - 1);
1275
1276 if (conf_name == L"new_command") {
1277 ADDITIONAL_COMMANDS[new_command_name] = command_value;
1278 }
1279 if (conf_name == L"new_macro") {
1280 ADDITIONAL_MACROS[new_command_name] = command_value;
1281 }
1282 if (conf_name == L"new_js_command" || conf_name == L"new_async_js_command") {
1283 QString qcommand_value = QString::fromStdWString(command_value);
1284 QStringList parts = qcommand_value.split("::");
1285 bool has_entry_point = parts.size() >= 2;
1286
1287 if (conf_name == L"new_js_command") {
1288 if (has_entry_point) {
1289 ADDITIONAL_JAVASCRIPT_COMMANDS[new_command_name] = JsCommandInfo{
1290 file_path.get_path(), parts.first().toStdWString(), parts.last().toStdWString()};
1291 }
1292 else {
1293 ADDITIONAL_JAVASCRIPT_COMMANDS[new_command_name] = JsCommandInfo{
1294 file_path.get_path(), command_value, {} };
1295 }
1296 }
1297 if (conf_name == L"new_async_js_command") {
1298 if (has_entry_point) {
1299 ADDITIONAL_ASYNC_JAVASCRIPT_COMMANDS[new_command_name] = JsCommandInfo{
1300 file_path.get_path(), parts.first().toStdWString(), parts.last().toStdWString()};
1301 }
1302 else {
1303 ADDITIONAL_ASYNC_JAVASCRIPT_COMMANDS[new_command_name] = JsCommandInfo{
1304 file_path.get_path(), command_value, {} };
1305 }
1306 }
1307 }
1308 }
1309 }
1310 }
1311 else {
1312
1313 Config* conf = get_mut_config_with_name(conf_name);
1314 if (conf == nullptr) {
1315 std::wcout << L"Error: " << conf_name << L" is not a valid configuration name\n";
1316 continue;
1317 }
1318
1319 std::wstring config_value;
1320 std::getline(ss, config_value);
1321
1322 std::wstringstream config_value_stream(config_value);
1323
1324 if (is_auto) {
1325 conf->is_auto = true;
1326 }
1327
1328 if ((conf != nullptr) && (conf->validator != nullptr)) {
1329 if (!conf->validator(config_value)) {
1330 std::wcout << L"Error in config file " << file_path.get_path() << L" at line " << line_number << L" : " << line << L"\n";
1331 continue;
1332 }
1333 }
1334
1335 if (conf) {
1336 bool changed = false;
1337 auto deserialization_result = conf->deserialize(config_value_stream, conf->value, &changed);
1338 if (deserialization_result != nullptr) {
1339 conf->value = deserialization_result;
1340 }
1341 else {
1342 std::wcout << L"Error in config file " << file_path.get_path() << L" at line " << line_number << L" : " << line << L"\n";
1343 }
1344 if (changed && (changed_config_names != nullptr)) {
1345 changed_config_names->push_back(utf8_encode(conf->name));
1346 }
1347 }
1348 }
1349
1350 }
1351 default_file.close();
1352}
