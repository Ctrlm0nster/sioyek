100std::wstring strip_uri(std::wstring pdf_file_name, std::optional<int>* out_page) {
101
102 QString pdf_file_name_qstring = QString::fromStdWString(pdf_file_name);
103 if (pdf_file_name_qstring.size() > 2 && pdf_file_name_qstring[0] == '"' && pdf_file_name_qstring[pdf_file_name_qstring.size() - 1] == '"') {
104 pdf_file_name_qstring = pdf_file_name_qstring.mid(1, pdf_file_name_qstring.size() - 2);
105 }
106
107 if (pdf_file_name_qstring.startsWith("file:")) {
108 QUrl url = QUrl::fromUserInput(pdf_file_name_qstring, QString(), QUrl::AssumeLocalFile);
109 if (!url.isLocalFile()) return pdf_file_name_qstring.toStdWString();
110
111 if (out_page) {
112 QUrlQuery fragment_query{url.fragment()};
113
114 bool ok;
115 int page = fragment_query.queryItemValue("page", QUrl::FullyDecoded).toInt(&ok);
116 if (ok) *out_page = page - 1;
117 }
118
119 return url.path().toStdWString();
120 }
121 else{
122 return pdf_file_name_qstring.toStdWString();
123 }
124}
