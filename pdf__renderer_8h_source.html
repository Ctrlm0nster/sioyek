139 static int find_closest_index(const std::vector<RenderResponse>& responses,
140 const RenderRequest& request) {
141 if (!std::isfinite(request.zoom_level) || request.zoom_level <= 0 ||
142 !std::isfinite(request.display_scale) || request.display_scale <= 0)
143 return -1;
144 float distance = std::numeric_limits<float>::infinity();
145 int best = -1;
146 for (size_t i = 0; i < responses.size(); ++i) {
147 const auto& response = responses[i];
148 const auto& cached = response.request;
149 if (response.pending || response.invalid || !response.texture ||
150 response.width <= 0 || response.height <= 0 ||
151 !std::isfinite(cached.zoom_level) || cached.zoom_level <= 0 ||
152 cached.path != request.path || cached.page != request.page ||
153 cached.slice_index != request.slice_index ||
154 cached.num_h_slices != request.num_h_slices || cached.num_v_slices != request.num_v_slices ||
155 cached.display_scale != request.display_scale || cached.gamma != request.gamma ||
156 cached.should_render_annotations != request.should_render_annotations)
157 continue;
158 const float candidate_distance = std::abs(cached.zoom_level - request.zoom_level);
159 if (candidate_distance < distance ||
160 (candidate_distance == distance && (best < 0 || cached.zoom_level > responses[best].request.zoom_level))) {
161 distance = candidate_distance;
162 best = static_cast<int>(i);
163 }
164 }
165 return best;
166 }
